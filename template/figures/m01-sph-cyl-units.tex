% Ch1_3, slides 15-16: the unit vectors a_R, a_theta and a_r, a_z at a point at the angle theta
% from the z-axis
\begin{tikzpicture}[ELoblique,scale=1]
  \coordinate (O) at (0,0,0);
  \def\t{38}\def\p{65}\def\R{2.6}
  \coordinate (P) at ({\R*sin(\t)*cos(\p)},{\R*sin(\t)*sin(\p)},{\R*cos(\t)});
  \draw[ELax] (O)--(2.4,0,0) node[ELlab,anchor=north east]{$x$};
  \draw[ELax] (O)--(0,2.8,0) node[ELlab,anchor=west]{$y$};
  \draw[ELax] (O)--(0,0,2.6) node[ELlab,anchor=south]{$z$};
  \draw[ELdash] (O)--(P);
  \draw[ELthin] plot[domain=0:\t,samples=20] ({0.8*sin(\x)*cos(\p)},{0.8*sin(\x)*sin(\p)},{0.8*cos(\x)});
  \node[ELlab,anchor=south west] at ({0.95*sin(\t/2)*cos(\p)},{0.95*sin(\t/2)*sin(\p)},{0.95*cos(\t/2)}) {$\theta$};
  \fill[ELink] (P) circle (1.5pt);
  \def\u{0.9}
  \draw[ELvec] (P)--++(0,0,\u) node[ELlabs,anchor=south]{$\uvec{z}$};
  \draw[ELvec] (P)--++({\u*cos(\p)},{\u*sin(\p)},0) node[ELlabs,anchor=west]{$\uvec{r}$};
  \draw[ELvec2] (P)--++({\u*sin(\t)*cos(\p)},{\u*sin(\t)*sin(\p)},{\u*cos(\t)}) node[ELlabs,anchor=south west]{$\uvec{R}$};
  \draw[ELvec2] (P)--++({\u*cos(\t)*cos(\p)},{\u*cos(\t)*sin(\p)},{-\u*sin(\t)}) node[ELlabs,anchor=north west]{$\uvec{\theta}$};
\end{tikzpicture}
