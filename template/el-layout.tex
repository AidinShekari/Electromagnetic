% ---------- displayed formulas ----------
% A displayed formula is centred and always left to right.  The Markdown filter marks the
% places where it may be broken (before a top-level =, \Rightarrow ..., after \qquad); lines
% are filled greedily by their measured width and continuation lines are indented.
\newtoks\ELdcur
\newbox\ELdb \newbox\ELdvb
\newcount\ELdn \newcount\ELdlines
\newdimen\ELdlim
\newcommand{\ELdline}[1]{\hbox{\ifELrtl\beginL\fi$\displaystyle#1$\ifELrtl\endL\fi}}
\def\ELdflush{%
  \setbox\ELdvb\vbox{\offinterlineskip\unvbox\ELdvb
    \ifnum\ELdlines>0 \vskip0.55em\fi
    \hbox{\ifnum\ELdlines>0 \hskip2em\fi\ELdline{\the\ELdcur}}}%
  \advance\ELdlines1 \ELdcur={}\ELdn=0 }
\long\def\ELdseg#1{%
  \ifnum\ELdn>0
    \setbox\ELdb\hbox{$\displaystyle\the\ELdcur #1$}%
    \ifdim\wd\ELdb>\ELdlim \ELdflush \fi
  \fi
  \ELdcur=\expandafter{\the\ELdcur #1}\advance\ELdn1 }
\long\def\ELdproc#1\ELbr#2\ELdend{%
  \ELdseg{#1}%
  \ifblank{#2}{}{\ELdproc#2\ELdend}}
\let\ELbr\relax
\newcommand{\ELdisp}[1]{\par\nopagebreak[4]\vspace{0.3em}%
  \begingroup\ELdcur={}\ELdn=0 \ELdlines=0 \setbox\ELdvb\vbox{}%
  \ELdlim=\dimexpr\linewidth-2em\relax
  \ELdproc#1\ELbr\ELdend
  \ifnum\ELdlines>0 \ELdflush \else
    \setbox\ELdvb\hbox{\ELdline{\the\ELdcur}}\fi
  \ifdim\wd\ELdvb>\linewidth\setbox\ELdvb\hbox{\resizebox{\linewidth}{!}{\box\ELdvb}}\fi
  \noindent\hfil\box\ELdvb\hfil\par\endgroup\nopagebreak[2]\vspace{0.1em}}

% ---------- logical blocks: never split a block that fits on a page ----------
% A heading followed by a box (or a unit) is held back until that block knows how much room it
% needs; heading and block then move together.  A block (a box, or a "unit": a paragraph with
% its equations, a figure with its lead-in) is typeset once into a box to measure it.  If it
% fits on a page it is kept whole, moving to the next page when it does not fit in what is left
% of the current one.  A block longer than a page stays breakable.
\newif\ifELhold
\def\ELpending{}\newcount\ELpendn
\newcommand{\ELhold}{\global\ELholdtrue}
\newcommand{\ELqueue}[1]{\gappto\ELpending{#1}\global\advance\ELpendn by 1\relax}
\newcommand{\ELsection}[2]{\ifELhold\ELqueue{\ELsectionreal{#1}{#2}}\else\ELsectionreal{#1}{#2}\fi}
\newcommand{\ELsubsection}[2]{\ifELhold\ELqueue{\ELsubsectionreal{#1}{#2}}\else\ELsubsectionreal{#1}{#2}\fi}
\newcommand{\ELflush}{\par\ifnum\ELpendn>0
  \begingroup\global\ELholdfalse\ELpending\endgroup
  \global\let\ELpending\empty\global\ELpendn=0 \fi\global\ELholdfalse}
\newdimen\ELpendht
\newdimen\ELboxht
\newdimen\ELkeepmax
\newcount\ELsvitem \newcount\ELsvfig \newcount\ELsvtab
\newcommand{\ELsavecnt}{\ELsvitem=\value{ELitem}\relax\ELsvfig=\value{ELfig}\relax\ELsvtab=\value{ELtab}\relax}
\newcommand{\ELrestorecnt}{\setcounter{ELitem}{\ELsvitem}\setcounter{ELfig}{\ELsvfig}\setcounter{ELtab}{\ELsvtab}}
\newif\ifELmeasuring
% #1 = width of the measured text, #2 = what is measured, #3 = extra height (padding)
\newcommand{\ELmeasureraw}[3]{\ELsavecnt
  \setbox0\vbox{\ELmeasuringtrue\hsize#1\linewidth\hsize\columnwidth\hsize\parindent0pt\relax
    \setlength{\parskip}{0.45em}#2\par}%
  \ELrestorecnt
  \ELboxht=\dimexpr\ht0+\dp0+#3\relax}
\newif\ifELnoplace
\newcommand{\ELplace}{% decides the room, then places the held headings
  \ifELnoplace\def\ELboxopt{unbreakable}\else
  \ELkeepmax=\dimexpr\textheight-1.5\baselineskip\relax
  \ELpendht=\dimexpr\ELpendn\baselineskip*3\relax
  \ifdim\ELboxht<\ELkeepmax
    \Needspace*{\dimexpr\ELboxht+\ELpendht+0.6\baselineskip\relax}\def\ELboxopt{unbreakable}%
  \else\Needspace*{\dimexpr7\baselineskip+\ELpendht\relax}\def\ELboxopt{}\fi
  \ELflush\fi}
% a statement and the proof that follows it form one block when together they fit on a page
\NewEnviron{ELjoin}{%
  \ELnoplacetrue\ELmeasureraw{\linewidth}{\BODY}{0pt}\ELnoplacefalse
  \ifdim\ELboxht<\dimexpr\textheight-1.5\baselineskip\relax
    \ELplace\ELnoplacetrue\BODY\ELnoplacefalse
  \else\ELplace\BODY\fi\par}
% ordinary text that belongs together (generated by the filter)
\NewEnviron{ELunit}{%
  \ELmeasureraw{\linewidth}{\BODY}{0pt}%
  \ELplace
  \BODY\par}
