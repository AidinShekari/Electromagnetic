% ============================================================
%  Electromagnetics — the environment family
%
%  One grammar for every block: a strong rule across the top of the block, a label tab with a
%  cut corner sitting on that rule at the leading edge (icon, name, number), a very light tint,
%  and a thin closing rule.  No side frames.  The categories differ by colour, icon and one
%  detail each:
%    definition  blue,   ≜ icon
%    theorem     navy,   ∇ icon      lemma (dipole), proposition (⊙), corollary (arrow)
%    important   gold,   spark icon,  closed thin frame on all sides
%    example     teal,   wave icon
%    exercise    copper, pen-nib icon, dashed closing rule
%    remark      slate,  info icon,   no tint
%    proof       navy,   turnstile,   no tab: a run-in heading, a thin rail at the leading
%                edge and the end-of-proof square (the only block that has one)
% ============================================================
% ---------- the label tab ----------
% the cut corner is at the trailing top corner of the tab (top right in English, top left in
% Persian), like the chamfer of a waveguide flange
\ifELrtl\def\ELtitleside{right}\def\ELtabx{-9pt}\else\def\ELtitleside{left}\def\ELtabx{9pt}\fi
\newlength{\ELcut}\setlength{\ELcut}{4pt}
\tcbset{ELtab/.style={enhanced,frame hidden,interior hidden,boxrule=0pt,sharp corners,
  left=5pt,right=6pt,top=1.6pt,bottom=1.4pt,
  colupper=white,fontupper=\ELheadfont\bfseries\small,
  overlay={\fill[#1] \ELcutpath{frame}{\ELcut};}}}
% the top and closing rules of a box (drawn per fragment when the box breaks)
\newcommand{\ELtoprule}[1]{\draw[#1,line width=1pt] (frame.north west)--(frame.north east);}
\newcommand{\ELbotrule}[2]{\draw[#1!45,line width=0.5pt,#2] (frame.south west)--(frame.south east);}

\tcbset{ELbase/.style n args={3}{% colour, tint, closing-rule style
  enhanced,breakable,sharp corners,frame hidden,boxrule=0pt,colback=#2,
  left=10pt,right=10pt,top=6pt,bottom=7pt,before skip=1em,after skip=1em,
  before upper={\parindent0pt\setlength{\parskip}{0.45em}},
  fonttitle=\ELheadfont\bfseries\small,coltitle=white,
  attach boxed title to top \ELtitleside={yshift=-0.5pt,xshift=\ELtabx,yshifttext=-1pt},
  boxed title style={ELtab=#1},
  varwidth boxed title*=-30pt,
  overlay unbroken={\ELtoprule{#1}\ELbotrule{#1}{#3}},
  overlay first={\ELtoprule{#1}},
  overlay middle={},
  overlay last={\ELbotrule{#1}{#3}}}}

% Numbers: a block carries the number printed in the instructor's notes (\ELnextnum{1-3} before
% it, written by the filter from {.example number="1-3"}); a block without one is unnumbered.
\def\ELpendingnum{}
\newcommand{\ELnextnum}[1]{\gdef\ELpendingnum{#1}}
\newcommand{\ELtakenum}{\ifx\ELpendingnum\empty\def\ELthisnum{}\else
  \edef\ELthisnum{\noexpand\ELnumc{\ELpendingnum}}\global\let\ELpendingnum\empty\fi}
% the text of a tab: icon, label, number, and the title (if any) after a thin stroke
\newcommand{\ELtabtext}[4]{% icon, label word, number (may be empty), title (may be empty)
  \ELicon{#1}{white}\hspace{0.4em}#2\if\relax\detokenize{#3}\relax\else~#3\fi
  \if\relax\detokenize{#4}\relax\else\hspace{0.5em}{\color{white!60}\rule[-0.2em]{0.5pt}{0.95em}}\hspace{0.5em}%
    {\mdseries #4}\fi}
% #1 env, #2 colour, #3 tint, #4 label word, #5 numbered (1/0), #6 icon, #7 closing-rule style
\newcommand{\ELmakebox}[7]{%
  \NewEnviron{#1}[1]{%
    \ELtakenum % (before measuring: a box nested in the body must not take this number)
    \ELmeasureraw{\dimexpr\linewidth-20pt\relax}{\BODY}{2.6em}%
    \ELplace
    \begin{tcolorbox}[ELbase={#2}{#3}{#7},\ELboxopt,title={\ELtabtext{#6}{#4}{\ELthisnum}{##1}}]%
    \BODY\end{tcolorbox}}}
\ELmakebox{ELdefinition}{ELcDefinition}{ELcDefinitionT}{\ELlangDefinition}{1}{defeq}{}
\ELmakebox{ELtheorem}{ELcTheorem}{ELcTheoremT}{\ELlangTheorem}{1}{nabla}{}
\ELmakebox{ELlemma}{ELcTheorem}{ELcTheoremT}{\ELlangLemma}{1}{dipole}{}
\ELmakebox{ELproposition}{ELcTheorem}{ELcTheoremT}{\ELlangProposition}{1}{odot}{}
\ELmakebox{ELcorollary}{ELcTheorem}{ELcTheoremT}{\ELlangCorollary}{1}{arrow}{}
\ELmakebox{ELexample}{ELcExample}{ELcExampleT}{\ELlangExample}{1}{wave}{}
\ELmakebox{ELexercise}{ELcExercise}{ELcExerciseT}{\ELlangExercise}{1}{nib}{dashed}
\ELmakebox{ELremark}{ELcRemark}{white}{\ELlangRemark}{0}{info}{}

% the important result: the same grammar, closed by a thin gold frame on every side
\newcommand{\ELimpframe}[2]{\draw[ELcImportant!70,line width=0.6pt] (frame.#2 west) rectangle (frame.#1 east);}
\NewEnviron{ELimportant}[1]{%
  \ELtakenum
  \ELmeasureraw{\dimexpr\linewidth-22pt\relax}{\BODY}{2.8em}%
  \ELplace
  \begin{tcolorbox}[ELbase={ELcImportant}{ELcImportantT}{},\ELboxopt,
    top=7pt,
    overlay unbroken={\ELimpframe{north}{south}\ELtoprule{ELcImportant}},
    overlay first={\ELimpframe{north}{south}\ELtoprule{ELcImportant}},
    overlay middle={\ELimpframe{north}{south}},
    overlay last={\ELimpframe{north}{south}},
    title={\ELtabtext{bolt}{\ELlangImportant}{\ELthisnum}{#1}}]%
  \BODY\end{tcolorbox}}

% ---------- proof ----------
% a run-in heading, a thin rail at the leading edge, and the end-of-proof square at the end
% of the last line
\newcommand{\ELqed}{\par\nobreak\vspace{-0.2em}\nopagebreak{\hfill\color{ELcProof}\ensuremath{\blacksquare}}\par}
\newcommand{\ELproofhead}{\noindent\ELicon{turnstile}{ELcProof}\hspace{0.4em}%
  {\ELheadfont\bfseries\color{ELcProof}\ELlangProof.}\par\nobreak\vspace{-0.1em}}
\ifELrtl\tcbset{ELproofpad/.style={left=4pt,right=12pt}}\else\tcbset{ELproofpad/.style={left=12pt,right=4pt}}\fi
\NewEnviron{ELproof}[1]{%
  \ELmeasureraw{\dimexpr\linewidth-14pt\relax}{\ELproofhead\BODY\ELqed}{1.6em}%
  \ELplace
  \begin{tcolorbox}[enhanced,breakable,\ELboxopt,sharp corners,frame hidden,interior hidden,boxrule=0pt,
    ELproofpad,top=3pt,bottom=3pt,
    before skip=0.7em,after skip=0.9em,
    before upper={\parindent0pt\setlength{\parskip}{0.45em}},
    overlay={\draw[ELcProof!35,line width=0.9pt]
      ([xshift=\ELsgn*2pt]frame.north \ELstart)--([xshift=\ELsgn*2pt]frame.south \ELstart);}]%
  \ELproofhead\BODY\ELqed\end{tcolorbox}}

% a solution inside an example or an exercise: a run-in heading with a small arrow tip
\newenvironment{ELsolution}{\par\noindent\ELsubtip\hspace{0.4em}%
  {\ELheadfont\bfseries\color{ELcExample}\ELlangSolution.}\ }{\par}
